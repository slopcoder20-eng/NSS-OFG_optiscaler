#version 450
#extension GL_KHR_cooperative_matrix : enable
#extension GL_KHR_shader_subgroup_basic : enable
#extension GL_KHR_shader_subgroup_ballot : enable
#extension GL_KHR_memory_scope_semantics : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int8 : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int32 : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int64 : enable

// Workgroup = exactly one subgroup (32 on wave32, 64 on wave64): coopmat ops need every lane of the subgroup.
layout(local_size_x_id = 0, local_size_y = 1, local_size_z = 1) in;
layout(constant_id = 0) const uint kSubgroupSize = 32;

// Push Constants (52 bytes)
layout(push_constant) uniform PushConstants {
    uint in_width;
    uint in_height;
    uint out_width;
    uint out_height;
    uint in_channels;
    uint out_channels;
    int pad_y;
    int pad_x;
    int stride_y;
    int stride_x;
    int out_zp;
    int has_table;
    uint in1_channels; // 0 for single-input, 16 for dual-input fused concat
    uint upsample;     // 1: input is read through a fused nearest 2x upsample (source is in_width/2 x in_height/2)
} pc;

layout(std430, set = 0, binding = 0) readonly buffer InputBuffer {
    uvec4 in_data[];
};

layout(std430, set = 0, binding = 1) writeonly buffer OutputBuffer {
    uint out_data[];
};

layout(std430, set = 0, binding = 2) readonly buffer WeightBuffer {
    int8_t weight_data[];
};

layout(std430, set = 0, binding = 3) readonly buffer BiasBuffer {
    int32_t bias_data[];
};

layout(std430, set = 0, binding = 4) readonly buffer MultiplierBuffer {
    int32_t multiplier_data[];
};

layout(std430, set = 0, binding = 5) readonly buffer ShiftBuffer {
    int32_t shift_data[];
};

layout(std430, set = 0, binding = 6) readonly buffer TableBuffer {
    int8_t table_data[];
};

layout(std430, set = 0, binding = 7) readonly buffer InputBuffer2 {
    uvec4 in_data2[];
};

uint inWord(uint w)  { return in_data[w >> 2][w & 3u]; }

// PT pixel tiles (16 pixels each) share every weight matrix B: 64 pixels x 16 output channels per subgroup.
#define PT 4
#define PIX (PT * 16)

shared uvec4   lds_tile_a[PT * 16];       // PT tiles x 16 pixel rows; one uvec4 = 16 int8 channels (128-bit LDS store)
shared int32_t lds_tile_c[256];           // one 16x16 int32 accumulator tile at a time
shared int32_t lds_bias[16];
shared int32_t lds_mult[16];
shared int32_t lds_shift[16];

void main() {
    const uint tile_p  = gl_WorkGroupID.x * PIX;
    const uint tile_oc = gl_WorkGroupID.y * 16;
    const uint lane_id = gl_SubgroupInvocationID;
    const uint SG = kSubgroupSize;
    const uint total_out_pixels = pc.out_width * pc.out_height;

    if (lane_id < 16) {
        uint cur_oc = tile_oc + lane_id;
        if (cur_oc < pc.out_channels) {
            lds_bias[lane_id]  = bias_data[cur_oc];
            lds_mult[lane_id]  = multiplier_data[cur_oc];
            lds_shift[lane_id] = shift_data[cur_oc];
        } else {
            lds_bias[lane_id]  = 0;
            lds_mult[lane_id]  = 0;
            lds_shift[lane_id] = 0;
        }
    }

    // Each lane owns pixels lane, lane+SG, ... of the PIX-pixel block (1 pixel on wave64, 2 on wave32).
    const uint PPL = PIX / SG;
    uint  my_oy[2];
    uint  my_ox[2];
    bool  my_ok[2];
    for (uint k = 0; k < PPL; ++k) {
        uint p = tile_p + lane_id + k * SG;
        my_ok[k] = (p < total_out_pixels);
        my_oy[k] = p / pc.out_width;
        my_ox[k] = p % pc.out_width;
    }

    coopmat<int32_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseAccumulator> matC[PT];
    for (uint t = 0; t < PT; ++t)
        matC[t] = coopmat<int32_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseAccumulator>(0);

    const uint ic_tiles = (pc.in_channels + 15) / 16;
    const uint oc_tiles = (pc.out_channels + 15) / 16;

    // Layers whose input is not "16 channels x contiguous pixels" go through LDS staging:
    // 12-channel first layer, fused-concat inputs, fused nearest-2x upsample.
    const bool special = (pc.in_channels == 12u) || (pc.in1_channels > 0u) || (pc.upsample != 0u);

    // Tile geometry for the direct path (all uniform across the subgroup)
    uint t_oy[PT];
    uint t_ox[PT];
    bool t_ok[PT];
    for (uint t = 0; t < PT; ++t) {
        uint p0 = tile_p + t * 16u;
        t_oy[t] = p0 / pc.out_width;
        t_ox[t] = p0 % pc.out_width;
        t_ok[t] = (p0 + 15u < total_out_pixels) && (t_ox[t] + 16u <= pc.out_width);
    }

    subgroupBarrier();
    subgroupMemoryBarrierShared();

    for (int ky = 0; ky < 3; ++ky) {
        for (int kx = 0; kx < 3; ++kx) {
            for (uint ic_t = 0; ic_t < ic_tiles; ++ic_t) {
                uint weight_tile_idx = (((ky * 3 + kx) * ic_tiles + ic_t) * oc_tiles + gl_WorkGroupID.y) * 256;
                coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseB> matB;
                coopMatLoad(matB, weight_data, weight_tile_idx, 16, gl_CooperativeMatrixLayoutRowMajor);

                if (special) {
                    // ---- collective staging of the whole PIX block (padding = raw zero-point -128) ----
                    for (uint k = 0; k < PPL; ++k) {
                        uvec4 val_vec = uvec4(0x80808080u);
                        if (my_ok[k]) {
                            int iy = int(my_oy[k]) * pc.stride_y + ky - pc.pad_y;
                            int ix = int(my_ox[k]) * pc.stride_x + kx - pc.pad_x;
                            if (iy >= 0 && iy < int(pc.in_height) && ix >= 0 && ix < int(pc.in_width)) {
                                uint p_in = (pc.upsample != 0u) ? ((uint(iy) >> 1) * (pc.in_width >> 1) + (uint(ix) >> 1)) : (uint(iy) * pc.in_width + uint(ix));
                                if (pc.in_channels == 12) {
                                    uint b = p_in * 3u;
                                    val_vec = uvec4(inWord(b), inWord(b + 1u), inWord(b + 2u), 0u);
                                } else if (pc.in1_channels > 0) {
                                    if (ic_t == 0)
                                        val_vec = in_data[p_in];
                                    else
                                        val_vec = in_data2[p_in * 2u + (ic_t - 1u)];
                                } else {
                                    val_vec = in_data[p_in * ic_tiles + ic_t];
                                }
                            }
                        }
                        lds_tile_a[lane_id + k * SG] = val_vec;
                    }
                    subgroupBarrier();
                    subgroupMemoryBarrierShared();
                    for (uint t = 0; t < PT; ++t) {
                        coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseA> matA;
                        coopMatLoad(matA, lds_tile_a, t * 16, 1, gl_CooperativeMatrixLayoutRowMajor);
                        matC[t] = coopMatMulAdd(matA, matB, matC[t]);
                    }
                    subgroupBarrier();
                    subgroupMemoryBarrierShared();
                } else {
                    // ---- direct path: A rows come straight from the (uvec4-typed) input buffer ----
                    for (uint t = 0; t < PT; ++t) {
                        coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseA> matA;
                        bool direct = false;
                        int iy = 0, ixs = 0;
                        if (t_ok[t]) {
                            iy  = int(t_oy[t]) * pc.stride_y + ky - pc.pad_y;
                            ixs = int(t_ox[t]) * pc.stride_x + kx - pc.pad_x;
                            int ixe = ixs + 15 * pc.stride_x;
                            direct = (iy >= 0) && (iy < int(pc.in_height)) && (ixs >= 0) && (ixe < int(pc.in_width));
                        }
                        if (direct) {
                            uint base = (uint(iy) * pc.in_width + uint(ixs)) * ic_tiles + ic_t;   // uvec4 units
                            coopMatLoad(matA, in_data, base, uint(pc.stride_x) * ic_tiles, gl_CooperativeMatrixLayoutRowMajor);
                        } else {
                            // border / partial tile: stage this tile's 16 pixels
                            if (lane_id < 16u) {
                                uint p = tile_p + t * 16u + lane_id;
                                uvec4 val_vec = uvec4(0x80808080u);
                                if (p < total_out_pixels) {
                                    int y = int(p / pc.out_width) * pc.stride_y + ky - pc.pad_y;
                                    int x = int(p % pc.out_width) * pc.stride_x + kx - pc.pad_x;
                                    if (y >= 0 && y < int(pc.in_height) && x >= 0 && x < int(pc.in_width))
                                        val_vec = in_data[(uint(y) * pc.in_width + uint(x)) * ic_tiles + ic_t];
                                }
                                lds_tile_a[t * 16u + lane_id] = val_vec;
                            }
                            subgroupBarrier();
                            subgroupMemoryBarrierShared();
                            coopMatLoad(matA, lds_tile_a, t * 16u, 1, gl_CooperativeMatrixLayoutRowMajor);
                            subgroupBarrier();
                            subgroupMemoryBarrierShared();
                        }
                        matC[t] = coopMatMulAdd(matA, matB, matC[t]);
                    }
                }
            }
        }
    }

    const uint out_oc_remain = (pc.out_channels > tile_oc) ? (pc.out_channels - tile_oc) : 0u;
    const uint valid_oc = min(out_oc_remain, 16u);

    for (uint t = 0; t < PT; ++t) {
        coopMatStore(matC[t], lds_tile_c, 0, 16, gl_CooperativeMatrixLayoutRowMajor);
        subgroupBarrier();
                subgroupMemoryBarrierShared();

        // 16 pixels x 4 words per tile, spread over all lanes
        for (uint it = lane_id; it < 64u; it += SG) {
            uint rr = it >> 2;          // pixel within the tile
            uint w  = it & 3u;          // output word (4 channels)
            uint p  = tile_p + t * 16u + rr;
            if (p >= total_out_pixels || w * 4u >= valid_oc) continue;

            uint out_base_u32 = p * (pc.out_channels / 4) + (tile_oc / 4) + w;
            uint packed_word = 0u;
            for (uint j = 0; j < 4u; ++j) {
                uint c = w * 4u + j;
                if (c >= valid_oc) break;
                int32_t acc = lds_tile_c[rr * 16u + c] + lds_bias[c];
                int64_t prod = int64_t(acc) * int64_t(lds_mult[c]);
                int32_t shift = lds_shift[c];
                int64_t round_val = (shift > 0) ? (int64_t(1) << (shift - 1)) : int64_t(0);
                int32_t val = int32_t((prod + round_val) >> shift) + pc.out_zp;
                int8_t clamped = int8_t(clamp(val, -128, 127));
                if (pc.has_table != 0) clamped = table_data[uint(int(clamped) + 128)];
                packed_word |= (uint(clamped) & 0xFFu) << (8u * j);
            }
            out_data[out_base_u32] = packed_word;
        }
        subgroupBarrier();
                subgroupMemoryBarrierShared();
    }
}
