#version 450
#extension GL_KHR_cooperative_matrix : enable
#extension GL_KHR_shader_subgroup_basic : enable
#extension GL_KHR_shader_subgroup_ballot : enable
#extension GL_KHR_memory_scope_semantics : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int8 : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int32 : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int64 : enable

// Workgroup = exactly one subgroup (32 on wave32, 64 on wave64): coopmat ops need every lane of the subgroup.
layout(local_size_x_id = 0, local_size_y = 1, local_size_z = 1) in;
layout(constant_id = 0) const uint kSubgroupSize = 32;
layout(constant_id = 2) const uint kAblate = 0;   // profiling only: 1 = no global input loads, 2 = no MMA, 3 = no epilogue, 4 = epilogue math but no global store, 5 = raw store without requant math
layout(constant_id = 1) const uint kUseSlide = 1;   // 0: always use the per-tap staging path (A/B switch)

// Push Constants (52 bytes)
layout(push_constant) uniform PushConstants {
    uint in_width;
    uint in_height;
    uint out_width;
    uint out_height;
    uint in_channels;
    uint out_channels;
    int pad_y;
    int pad_x;
    int stride_y;
    int stride_x;
    int out_zp;
    int has_table;
    uint in1_channels; // 0 for single-input, 16 for dual-input fused concat
    uint upsample;     // 1: input is read through a fused nearest 2x upsample (source is in_width/2 x in_height/2)
} pc;

layout(std430, set = 0, binding = 0) readonly buffer InputBuffer {
    uvec4 in_data[];
};

layout(std430, set = 0, binding = 1) writeonly buffer OutputBuffer {
    uint out_data[];
};

layout(std430, set = 0, binding = 2) readonly buffer WeightBuffer {
    int8_t weight_data[];
};

layout(std430, set = 0, binding = 3) readonly buffer BiasBuffer {
    int32_t bias_data[];
};

layout(std430, set = 0, binding = 4) readonly buffer MultiplierBuffer {
    int32_t multiplier_data[];
};

layout(std430, set = 0, binding = 5) readonly buffer ShiftBuffer {
    int32_t shift_data[];
};

layout(std430, set = 0, binding = 6) readonly buffer TableBuffer {
    int8_t table_data[];
};

layout(std430, set = 0, binding = 7) readonly buffer InputBuffer2 {
    uvec4 in_data2[];
};

// Same memory as in_data, seen as scalar words: the 12-channel first layer has 12-byte pixels (3 words).
layout(std430, set = 0, binding = 0) readonly buffer InputBufferWords {
    uint in_words[];
};

uint inWord(uint w)  { return in_words[w]; }

// One input pixel's 16-channel group `ic_t` (handles the 12-channel input, fused concat and fused 2x upsample).
uvec4 loadInPixel(uint y, uint x, uint ic_t, uint ic_tiles) {
    uint p_in = (pc.upsample != 0u) ? ((y >> 1) * (pc.in_width >> 1) + (x >> 1)) : (y * pc.in_width + x);
    if (pc.in_channels == 12u) {
        uint b = p_in * 3u;
        return uvec4(inWord(b), inWord(b + 1u), inWord(b + 2u), 0u);
    }
    if (pc.in1_channels > 0u)
        return (ic_t == 0u) ? in_data[p_in] : in_data2[p_in * 2u + (ic_t - 1u)];
    return in_data[p_in * ic_tiles + ic_t];
}

// A subgroup computes PT pixel tiles (16 pixels each) x OCT output-channel tiles (16 channels each):
// weights (matB) are shared by the PT pixel tiles, the staged input (matA) by the OCT channel tiles.
#define PT 4
#define OCT 2
#define PIX (PT * 16)

shared uvec4   lds_row[132];              // sliding window: up to 63*2+3 = 129 input pixels x 16 channels
shared uvec4   lds_tile_a[PT * 16];       // fallback staging: PT tiles x 16 pixel rows
shared int32_t lds_tile_c[256];           // one 16x16 int32 accumulator tile at a time
shared int32_t lds_bias[16 * OCT];
shared int32_t lds_mult[16 * OCT];
shared int32_t lds_shift[16 * OCT];

void main() {
    const uint tile_p   = gl_WorkGroupID.x * PIX;
    const uint oct_base = gl_WorkGroupID.y * OCT;          // first output-channel tile of this workgroup
    const uint lane_id  = gl_SubgroupInvocationID;
    const uint SG = kSubgroupSize;
    const uint total_out_pixels = pc.out_width * pc.out_height;

    const uint ic_tiles = (pc.in_channels + 15) / 16;
    const uint oc_tiles = (pc.out_channels + 15) / 16;

    for (uint i = lane_id; i < 16u * OCT; i += SG) {
        uint cur_oc = oct_base * 16u + i;
        if (cur_oc < pc.out_channels) {
            lds_bias[i]  = bias_data[cur_oc];
            lds_mult[i]  = multiplier_data[cur_oc];
            lds_shift[i] = shift_data[cur_oc];
        } else {
            lds_bias[i]  = 0;
            lds_mult[i]  = 0;
            lds_shift[i] = 0;
        }
    }

    const uint PPL = PIX / SG;
    uint  my_oy[PT / 2];
    uint  my_ox[PT / 2];
    bool  my_ok[PT / 2];
    for (uint k = 0; k < PPL; ++k) {
        uint p = tile_p + lane_id + k * SG;
        my_ok[k] = (p < total_out_pixels);
        my_oy[k] = p / pc.out_width;
        my_ox[k] = p % pc.out_width;
    }

    coopmat<int32_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseAccumulator> matC[PT * OCT];
    for (uint i = 0; i < PT * OCT; ++i)
        matC[i] = coopmat<int32_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseAccumulator>(0);

    const bool oc1_valid = (oct_base + 1u) < oc_tiles;     // second channel tile exists (uniform)

    subgroupBarrier();
    subgroupMemoryBarrierShared();

    if (kUseSlide != 0u && (pc.out_width % PIX) == 0u) {
        // Sliding window: a 64-pixel block inside one output row needs one input row segment per ky;
        // the three kx taps read the same LDS rows at offset kx (stride 1 or 2).
        const uint oy0 = tile_p / pc.out_width;
        const uint x0  = tile_p % pc.out_width;
        const uint sx  = uint(pc.stride_x);
        const uint NP  = (PIX - 1u) * sx + 3u;
        for (int ky = 0; ky < 3; ++ky) {
            const int  iy    = int(oy0) * pc.stride_y + ky - pc.pad_y;
            const bool rowOk = (iy >= 0) && (iy < int(pc.in_height));
            for (uint ic_t = 0; ic_t < ic_tiles; ++ic_t) {
                for (uint j = lane_id; j < NP; j += SG) {
                    int x = int(x0 * sx) - pc.pad_x + int(j);
                    uvec4 v = uvec4(0x80808080u);
                    if (rowOk && x >= 0 && x < int(pc.in_width))
                        v = loadInPixel(uint(iy), uint(x), ic_t, ic_tiles);
                    lds_row[j] = v;
                }
                subgroupBarrier();
                subgroupMemoryBarrierShared();

                for (int kx = 0; kx < 3; ++kx) {
                    uint wbase = (((ky * 3 + kx) * ic_tiles + ic_t) * oc_tiles + oct_base) * 256u;
                    coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseB> matB0;
                    coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseB> matB1;
                    coopMatLoad(matB0, weight_data, wbase, 16, gl_CooperativeMatrixLayoutRowMajor);
                    if (oc1_valid)
                        coopMatLoad(matB1, weight_data, wbase + 256u, 16, gl_CooperativeMatrixLayoutRowMajor);
                    for (uint t = 0; t < PT; ++t) {
                        coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseA> matA;
                        coopMatLoad(matA, lds_row, t * 16u * sx + uint(kx), sx, gl_CooperativeMatrixLayoutRowMajor);
                        matC[t * OCT + 0] = coopMatMulAdd(matA, matB0, matC[t * OCT + 0]);
                        if (oc1_valid)
                            matC[t * OCT + 1] = coopMatMulAdd(matA, matB1, matC[t * OCT + 1]);
                    }
                }
                subgroupBarrier();
                subgroupMemoryBarrierShared();
            }
        }
    } else {
        for (int ky = 0; ky < 3; ++ky) {
            for (int kx = 0; kx < 3; ++kx) {
                for (uint ic_t = 0; ic_t < ic_tiles; ++ic_t) {
                    for (uint k = 0; k < PPL; ++k) {
                        uvec4 val_vec = uvec4(0x80808080u);
                        if (my_ok[k]) {
                            int iy = int(my_oy[k]) * pc.stride_y + ky - pc.pad_y;
                            int ix = int(my_ox[k]) * pc.stride_x + kx - pc.pad_x;
                            if (iy >= 0 && iy < int(pc.in_height) && ix >= 0 && ix < int(pc.in_width))
                                val_vec = loadInPixel(uint(iy), uint(ix), ic_t, ic_tiles);
                        }
                        lds_tile_a[lane_id + k * SG] = val_vec;
                    }
                    subgroupBarrier();
                    subgroupMemoryBarrierShared();

                    uint wbase = (((ky * 3 + kx) * ic_tiles + ic_t) * oc_tiles + oct_base) * 256u;
                    coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseB> matB0;
                    coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseB> matB1;
                    coopMatLoad(matB0, weight_data, wbase, 16, gl_CooperativeMatrixLayoutRowMajor);
                    if (oc1_valid)
                        coopMatLoad(matB1, weight_data, wbase + 256u, 16, gl_CooperativeMatrixLayoutRowMajor);
                    for (uint t = 0; t < PT; ++t) {
                        coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseA> matA;
                        coopMatLoad(matA, lds_tile_a, t * 16, 1, gl_CooperativeMatrixLayoutRowMajor);
                        matC[t * OCT + 0] = coopMatMulAdd(matA, matB0, matC[t * OCT + 0]);
                        if (oc1_valid)
                            matC[t * OCT + 1] = coopMatMulAdd(matA, matB1, matC[t * OCT + 1]);
                    }
                    subgroupBarrier();
                    subgroupMemoryBarrierShared();
                }
            }
        }
    }

    // ---- epilogue: per (channel tile, pixel tile): accumulators -> LDS -> requantize -> packed dword stores ----
    for (uint o = 0; o < OCT; ++o) {
        const uint tile_oc = (oct_base + o) * 16u;
        if (tile_oc >= pc.out_channels) break;
        const uint valid_oc = min(pc.out_channels - tile_oc, 16u);

        for (uint t = 0; t < PT; ++t) {
            coopMatStore(matC[t * OCT + o], lds_tile_c, 0, 16, gl_CooperativeMatrixLayoutRowMajor);
            subgroupBarrier();
            subgroupMemoryBarrierShared();

            for (uint it = lane_id; it < 64u; it += SG) {
                uint rr = it >> 2;          // pixel within the tile
                uint w  = it & 3u;          // output word (4 channels)
                uint p  = tile_p + t * 16u + rr;
                if (p >= total_out_pixels || w * 4u >= valid_oc) continue;

                uint out_base_u32 = p * (pc.out_channels / 4) + (tile_oc / 4) + w;
                uint packed_word = 0u;
                for (uint j = 0; j < 4u; ++j) {
                    uint c = w * 4u + j;
                    if (c >= valid_oc) break;
                    int32_t acc = lds_tile_c[rr * 16u + c] + lds_bias[o * 16u + c];
                    int32_t shift = lds_shift[o * 16u + c];
                    // Every rescale shift in both NSS networks is >= 35, so (acc*mult + 2^(shift-1)) >> shift
                    // only needs the upper 32 bits of the product (identical to the 64-bit form).
                    int32_t hi, lo;
                    imulExtended(acc, lds_mult[o * 16u + c], hi, lo);
                    int32_t val = ((hi + (1 << (shift - 33))) >> (shift - 32)) + pc.out_zp;
                    int8_t clamped = int8_t(clamp(val, -128, 127));
                    if (pc.has_table != 0) clamped = table_data[uint(int(clamped) + 128)];
                    packed_word |= (uint(clamped) & 0xFFu) << (8u * j);
                }
                out_data[out_base_u32] = packed_word;
            }
            subgroupBarrier();
            subgroupMemoryBarrierShared();
        }
    }
}
