#version 450
#extension GL_EXT_control_flow_attributes : enable
#extension GL_KHR_cooperative_matrix : enable
#extension GL_KHR_shader_subgroup_basic : enable
#extension GL_KHR_shader_subgroup_ballot : enable
#extension GL_KHR_memory_scope_semantics : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int8 : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int32 : enable
#extension GL_EXT_shader_explicit_arithmetic_types_int64 : enable

// Workgroup = exactly one subgroup (32 on wave32, 64 on wave64): coopmat ops need every lane of the subgroup.
layout(local_size_x_id = 0, local_size_y = 1, local_size_z = 1) in;
layout(constant_id = 0) const uint kSubgroupSize = 32;
layout(constant_id = 2) const uint kAblate = 0;   // profiling bitmask: 1 = no global input loads, 2 = no MMA, 4 = raw accumulator store instead of the epilogue, 8 = weights loaded once

// Push Constants (52 bytes)
layout(push_constant) uniform PushConstants {
    uint in_width;
    uint in_height;
    uint out_width;
    uint out_height;
    uint in_channels;
    uint out_channels;
    int pad_y;
    int pad_x;
    int stride_y;
    int stride_x;
    int out_zp;
    int has_table;
    uint in1_channels; // 0 for single-input, 16 for dual-input fused concat
    uint upsample;     // 1: input is read through a fused nearest 2x upsample (source is in_width/2 x in_height/2)
} pc;

layout(std430, set = 0, binding = 0) readonly buffer InputBuffer {
    uvec4 in_data[];
};

layout(std430, set = 0, binding = 1) writeonly buffer OutputBuffer {
    uint out_data[];
};

layout(std430, set = 0, binding = 2) readonly buffer WeightBuffer {
    int8_t weight_data[];
};

layout(std430, set = 0, binding = 2) readonly buffer WeightWordBuffer {
    uint weight_words[];
};

layout(std430, set = 0, binding = 3) readonly buffer BiasBuffer {
    int32_t bias_data[];
};

layout(std430, set = 0, binding = 4) readonly buffer MultiplierBuffer {
    int32_t multiplier_data[];
};

layout(std430, set = 0, binding = 5) readonly buffer ShiftBuffer {
    int32_t shift_data[];
};

layout(std430, set = 0, binding = 6) readonly buffer TableBuffer {
    int8_t table_data[];
};

layout(std430, set = 0, binding = 7) readonly buffer InputBuffer2 {
    uvec4 in_data2[];
};

// Same memory as in_data, seen as scalar words: the 12-channel first layer has 12-byte pixels (3 words).
layout(std430, set = 0, binding = 0) readonly buffer InputBufferWords {
    uint in_words[];
};

uint inWord(uint w)  { return in_words[w]; }

// One input pixel's 16-channel group `ic_t` (handles the 12-channel input, fused concat and fused 2x upsample).
uvec4 loadInPixel(uint y, uint x, uint ic_t, uint ic_tiles) {
    uint p_in = (pc.upsample != 0u) ? ((y >> 1) * (pc.in_width >> 1) + (x >> 1)) : (y * pc.in_width + x);
    if (pc.in_channels == 12u) {
        uint b = p_in * 3u;
        return uvec4(inWord(b), inWord(b + 1u), inWord(b + 2u), 0u);
    }
    if (pc.in1_channels > 0u)
        return (ic_t == 0u) ? in_data[p_in] : in_data2[p_in * 2u + (ic_t - 1u)];
    return in_data[p_in * ic_tiles + ic_t];
}

// A subgroup computes a 16x4 pixel block (4 pixel tiles = 4 consecutive rows, 16 columns each) x OCT output-channel tiles.
// Any output width works: columns >= out_width and rows >= out_height are masked. Weights (matB) are shared by the
// PT pixel tiles, the staged input rows (matA) by the OCT channel tiles.
#define PT 4
#define OCT 2
#define TW 16
#define NPMAX (15 * 2 + 3)      // input pixels needed per tile row (stride 2 worst case)

shared uvec4   lds_row[PT * NPMAX];       // sliding window: one input row segment per tile row
shared int32_t lds_tile_c[256];
shared int32_t lds_idx[256];           // one 16x16 int32 accumulator tile at a time
shared int32_t lds_bias[16 * OCT];
shared int32_t lds_mult[16 * OCT];
shared int32_t lds_shift[16 * OCT];

void main() {
    const uint oct_base = gl_WorkGroupID.y * OCT;          // first output-channel tile of this workgroup
    const uint lane_id  = gl_SubgroupInvocationID;
    const uint SG = kSubgroupSize;

    const uint nbx = (pc.out_width + TW - 1u) / TW;
    const uint bx  = gl_WorkGroupID.x % nbx;
    const uint by  = gl_WorkGroupID.x / nbx;
    const uint x0  = bx * TW;
    const uint y0  = by * PT;

    const uint ic_tiles = (pc.in_channels + 15) / 16;
    const uint oc_tiles = (pc.out_channels + 15) / 16;

    for (uint i = lane_id; i < 16u * OCT; i += SG) {
        uint cur_oc = oct_base * 16u + i;
        if (cur_oc < pc.out_channels) {
            lds_bias[i]  = bias_data[cur_oc];
            lds_mult[i]  = multiplier_data[cur_oc];
            lds_shift[i] = shift_data[cur_oc];
        } else {
            lds_bias[i]  = 0;
            lds_mult[i]  = 0;
            lds_shift[i] = 0;
        }
    }

    // Element-to-(row, col) map of the accumulator layout, learned by loading a matrix whose value is its own index.
    for (uint i = lane_id; i < 256u; i += SG)
        lds_idx[i] = int(i);
    subgroupBarrier();
    subgroupMemoryBarrierShared();
    coopmat<int32_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseAccumulator> idxM;
    coopMatLoad(idxM, lds_idx, 0, 16, gl_CooperativeMatrixLayoutRowMajor);
    const uint NEL = idxM.length();
    // If a lane only ever holds one column (true for the RDNA WMMA layout) its bias / multiplier / shift are loaded once per tile.
    const int col0 = idxM[0] & 15;
    bool colUni = true;
    [[unroll]] for (uint i = 1; i < NEL; ++i)
        colUni = colUni && ((idxM[i] & 15) == col0);

    coopmat<int32_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseAccumulator> matC[PT * OCT];
    for (uint i = 0; i < PT * OCT; ++i)
        matC[i] = coopmat<int32_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseAccumulator>(0);

    const bool oc1_valid = (oct_base + 1u) < oc_tiles;     // second channel tile exists (uniform)

    subgroupBarrier();
    subgroupMemoryBarrierShared();

    const uint sx = uint(pc.stride_x);
    const uint NP = (TW - 1u) * sx + 3u;
    // Per-lane staging slots: which (tile row, column) each lane loads never changes across the ky / ic loops,
    // so resolve the division, column bounds and row base once instead of in every staging phase.
    const uint NSLOT = (PT * NPMAX + kSubgroupSize - 1u) / kSubgroupSize;
    int  st_rowbase[NSLOT];
    int  st_x[NSLOT];
    uint st_dst[NSLOT];
    bool st_live[NSLOT];                               // slot addresses a valid LDS row and output row
    [[unroll]] for (uint sl = 0; sl < NSLOT; ++sl) {
        const uint j  = lane_id + sl * SG;
        const bool inr = j < PT * NP;
        const uint t  = inr ? j / NP : 0u;
        const uint jj = inr ? j - t * NP : 0u;
        st_live[sl]    = inr && ((y0 + t) < pc.out_height) && (int(x0 * sx) - pc.pad_x + int(jj) >= 0)
                         && (int(x0 * sx) - pc.pad_x + int(jj) < int(pc.in_width));
        st_rowbase[sl] = int(y0 + t) * pc.stride_y - pc.pad_y;
        st_x[sl]       = int(x0 * sx) - pc.pad_x + int(jj);
        st_dst[sl]     = inr ? (t * NP + jj) : 0u;
    }
    for (int ky = 0; ky < 3; ++ky) {
        for (uint ic_t = 0; ic_t < ic_tiles; ++ic_t) {
            [[unroll]] for (uint sl = 0; sl < NSLOT; ++sl) {
                const uint j = lane_id + sl * SG;
                if (j < PT * NP) {
                    const int iy = st_rowbase[sl] + ky;
                    uvec4 v = uvec4(0x80808080u);
                    if ((kAblate & 1u) != 0u) v = uvec4(0x01020304u + j);
                    else if (st_live[sl] && iy >= 0 && iy < int(pc.in_height))
                        v = loadInPixel(uint(iy), uint(st_x[sl]), ic_t, ic_tiles);
                    lds_row[st_dst[sl]] = v;
                }
            }
            subgroupBarrier();
            subgroupMemoryBarrierShared();

            for (int kx = 0; kx < 3; ++kx) {
                uint wbase = ((kAblate & 8u) != 0u) ? oct_base * 256u : (((ky * 3 + kx) * ic_tiles + ic_t) * oc_tiles + oct_base) * 256u;
                coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseB> matB0;
                coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseB> matB1;
                coopMatLoad(matB0, weight_words, wbase >> 2, 4, gl_CooperativeMatrixLayoutRowMajor);
                if (oc1_valid)
                    coopMatLoad(matB1, weight_words, (wbase + 256u) >> 2, 4, gl_CooperativeMatrixLayoutRowMajor);
                for (uint t = 0; t < PT; ++t) {
                    coopmat<int8_t, gl_ScopeSubgroup, 16, 16, gl_MatrixUseA> matA;
                    coopMatLoad(matA, lds_row, t * NP + uint(kx), sx, gl_CooperativeMatrixLayoutRowMajor);
                    if ((kAblate & 2u) == 0u) {
                    matC[t * OCT + 0] = coopMatMulAdd(matA, matB0, matC[t * OCT + 0]);
                    if (oc1_valid)
                        matC[t * OCT + 1] = coopMatMulAdd(matA, matB1, matC[t * OCT + 1]);
                    }
                }
            }
            subgroupBarrier();
            subgroupMemoryBarrierShared();
        }
    }

    // ---- epilogue: requantize directly on the accumulator registers, then repack channels through LDS ----
    for (uint o = 0; o < OCT; ++o) {
        const uint tile_oc = (oct_base + o) * 16u;
        if (tile_oc >= pc.out_channels) break;
        const uint valid_oc = min(pc.out_channels - tile_oc, 16u);

        for (uint t = 0; t < PT; ++t) {
            if (y0 + t >= pc.out_height) break;                 // uniform
            const uint kIdx = t * OCT + o;

            if (colUni) {
                if (uint(col0) < valid_oc) {
                    const int32_t bias  = lds_bias[o * 16u + uint(col0)];
                    const int32_t mult  = lds_mult[o * 16u + uint(col0)];
                    const int32_t shift = lds_shift[o * 16u + uint(col0)];
                    [[unroll]] for (uint i = 0; i < NEL; ++i) {
                        const int32_t pos = idxM[i];
                        // Every rescale shift in both NSS networks is >= 35, so (acc*mult + 2^(shift-1)) >> shift
                        // only needs the upper 32 bits of the product (identical to the 64-bit form).
                        int32_t hi, lo;
                        imulExtended(matC[kIdx][i] + bias, mult, hi, lo);
                        int32_t val = ((hi + (1 << (shift - 33))) >> (shift - 32)) + pc.out_zp;
                        int32_t clamped = clamp(val, -128, 127);
                        if (pc.has_table != 0) clamped = int32_t(table_data[uint(clamped + 128)]);
                        lds_tile_c[pos] = clamped;
                    }
                }
            } else {
                [[unroll]] for (uint i = 0; i < NEL; ++i) {
                    const int32_t pos = idxM[i];
                    const uint c = uint(pos & 15);
                    if (c < valid_oc) {
                        int32_t hi, lo;
                        const int32_t shift = lds_shift[o * 16u + c];
                        imulExtended(matC[kIdx][i] + lds_bias[o * 16u + c], lds_mult[o * 16u + c], hi, lo);
                        int32_t val = ((hi + (1 << (shift - 33))) >> (shift - 32)) + pc.out_zp;
                        int32_t clamped = clamp(val, -128, 127);
                        if (pc.has_table != 0) clamped = int32_t(table_data[uint(clamped + 128)]);
                        lds_tile_c[pos] = clamped;
                    }
                }
            }
            subgroupBarrier();
            subgroupMemoryBarrierShared();

            for (uint it = lane_id; it < 64u; it += SG) {
                uint rr = it >> 2;          // pixel column within the tile
                uint w  = it & 3u;          // output word (4 channels)
                uint ox = x0 + rr;
                if (ox >= pc.out_width || w * 4u >= valid_oc) continue;
                uint p  = (y0 + t) * pc.out_width + ox;
                uint out_base_u32 = p * (pc.out_channels / 4) + (tile_oc / 4) + w;
                uint packed_word = (uint(lds_tile_c[rr * 16u + w * 4u + 0u]) & 0xFFu)
                                 | ((uint(lds_tile_c[rr * 16u + w * 4u + 1u]) & 0xFFu) << 8)
                                 | ((uint(lds_tile_c[rr * 16u + w * 4u + 2u]) & 0xFFu) << 16)
                                 | ((uint(lds_tile_c[rr * 16u + w * 4u + 3u]) & 0xFFu) << 24);
                out_data[out_base_u32] = packed_word;
            }
            subgroupBarrier();
            subgroupMemoryBarrierShared();
        }
    }
}
